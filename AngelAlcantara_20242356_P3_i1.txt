#Eth1 DHCP: IP recibida del FG-Empresa en la VLAN 10
#IP obtenida: 10.23.56.10/25 - Gateway: 10.23.56.1

dhclient eth1
ip a
ip route

#Instalacion temporal de paquetes (por la red de Docker)
ip route replace default via 10.177.0.1 dev eth0
echo "nameserver 8.8.8.8" > /etc/resolv.conf
apt -o Acquire::ForceIPv4=true update
apt -o Acquire::ForceIPv4=true install -y curl openssh-client traceroute
ip route replace default via 10.23.56.1 dev eth1    # devolver ruta al FortiGate

#Pruebas
ping -c 4 10.23.56.1
ping -c 4 8.8.8.8                               # TTL=254
curl -k https://10.23.56.130                    # Sistema de Caja
curl http://10.23.56.131                        # Web Page Blocked! (Local URLfilter Block)
curl -k --max-time 5 https://10.23.56.131       # Connection timed out
ssh -o ConnectTimeout=5 angel@10.23.56.130      # Connection timed out
traceroute -n -T -p 443 10.23.56.130
#  1  10.23.56.1
#  2  10.23.56.130
